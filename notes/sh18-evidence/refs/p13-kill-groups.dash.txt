$ dash -i -m  < p13-kill-groups.keys
step 0 null
  err "$ "
step 1 "sh -c 'sleep 30 & sleep 31; wait' &"
  err "$ "

  err "$ "

  out "group-is-pid\n"
  err "$ "
step 5 "kill -- -$!"
  err "$ "

  err "[1] + Terminated                 sh -c \"sleep 30 & sleep 31; wait\"\n$ "
step 7 "jobs"
  err "$ "

  out "st=0\n"
  err "$ "

  err "$ "

  err "$ "

  err "[1] + Killed                     sleep 30\n$ "
step 12 "jobs"
  err "$ "
step 13 "kill -9 -999999"

  err "$ "
